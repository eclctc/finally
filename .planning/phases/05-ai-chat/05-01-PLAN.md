## Phase Goal

**As a** trader, **I want to** ask the AI to analyze and trade for me, **so that** I can act through conversation.

## Acceptance Criteria (this plan)

- [ ] The chat panel is open on load; the user can type a message and send it, and the reply appears in the transcript (CHAT-01, send half; Plans 05-02 and 05-05 complete it)
- [ ] Each trade and watchlist change the AI attempts shows inline under its reply with its status and reason, for executed and failed actions (CHAT-03)
- [ ] After a reply with successful actions, the portfolio and P&L history refresh without a reload, so the positions table shows the AI's trade (CHAT-05, portfolio half; Plan 05-03 adds the watchlist, Plan 05-04 the highlight)

export type ChatAction =
  | { type: "trade"; ticker: string; side: "buy" | "sell"; quantity: number; status: "ok"; price: number }
  | { type: "trade"; ticker: string; side: "buy" | "sell"; quantity: number; status: "failed"; error: string }
  | { type: "watchlist"; ticker: string; action: "add" | "remove"; status: "ok" }
  | { type: "watchlist"; ticker: string; action: "add" | "remove"; status: "failed"; error: string };
export interface ChatReply { message: string; actions: ChatAction[] }
export function describeAction(action: ChatAction): string;
// frontend/src/lib/useChat.ts
export interface ChatMessage { role: "user" | "assistant"; content: string; actions: ChatAction[] }
export interface ChatAttempt { text: string; error: string | null } // error null = in flight
export function useChat(onActionsApplied: (actions: ChatAction[]) => void): {
  messages: ChatMessage[]; attempt: ChatAttempt | null; pending: boolean;
  draft: string; setDraft: (text: string) => void; send: (text: string) => Promise<void>;
};
export type ChatState = ReturnType<typeof useChat>;
// frontend/src/components
export function ChatDock(props: { onActionsApplied: (actions: ChatAction[]) => void }): JSX.Element;
export function ChatTranscript(props: { chat: ChatState }): JSX.Element;
